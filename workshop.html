// Add ordinal suffix to day
function getDayWithSuffix(day) {
  if (day >= 11 && day <= 13) return `${day}th`;
  switch (day % 10) {
    case 1: return `${day}st`;
    case 2: return `${day}nd`;
    case 3: return `${day}rd`;
    default: return `${day}th`;
  }
}

// Convert "HH:mm" to "h:mm AM/PM"
function to12Hour(time24) {
  const [hours, minutes] = time24.split(":").map(Number);
  const suffix = hours >= 12 ? "PM" : "AM";
  const hour12 = hours % 12 || 12;
  return `${hour12}:${String(minutes).padStart(2, "0")} ${suffix}`;
}

const [hours, minutes] = inputData.workshopTime.split(":").map(Number);

// Base date = tomorrow
const d = new Date();
d.setDate(d.getDate() + 1);
d.setHours(hours, minutes, 0, 0);

const month = String(d.getMonth() + 1).padStart(2, "0");
const day = String(d.getDate()).padStart(2, "0");
const year = d.getFullYear();
const monthName = d.toLocaleString("en-US", { month: "long" });
const workshopDateFormatted = `${monthName} ${getDayWithSuffix(Number(day))}`;
const workshopTimePretty = to12Hour(inputData.workshopTime);

output = {
  result: {
    workshopDateTimeRaw: `${month}-${day}-${year} ${workshopTimePretty}`,
    workshopDateFormatted,
    workshopTimePretty
  }
};
